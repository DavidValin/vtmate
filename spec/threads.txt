%%{init: {'flowchart': {'curve': 'linear'}}}%%
flowchart LR

    %% Main thread
    Main(Main)

    %% Thread groups
    subgraph Audio[Audio Stack]
        Rec[Record thread]
        Conv[Conversation thread]
        TTS[TTS thread]
        Play[Playback thread]
    end
    subgraph Control[Control Stack]
        Key[Keyboard thread]
        UI[UI thread]
    end
    subgraph Daemon[Daemon mode only - replaces Key and UI]
        Ctl[Controller thread - global shortcuts, attached keys]
        Srv[IPC server thread - local socket, one reader/writer per attached client]
        Sink[UI sink thread - drains tx_ui, fans out to clients]
        Att[Attached vtmate - UI thread fed by the socket, keys forwarded]
    end

    %% Global stop propagation
    Main -->|stop_all_tx| Audio
    Main -->|stop_all_tx| Control

    %% Record -> Conversation
    Rec -->|tx_utt - microphone audio| Conv
    Rec -->|tx_utt - recognized utterance audio| Conv
    Rec -->|tx_ui - user_interrupt_show| UI

    %% Conversation -> others
    Conv -->|tx_tts - TTS sentence| TTS
    Conv -->|tx_play - speech audio| Play
    Conv -->|tx_ui - conversation chunks & 🧠 Memory saved| UI
    Conv -->|stop_all_tx - global stop| Main

    %% TTS -> Playback
    TTS -->|tx_play - synthetized audio chunks| Play
    TTS -->|stop_play_tx - interruption| Play

    %% Playback interactions
    Play -->|stop_play_rx - stop playback| Play
    Play -->|stop_all_rx - global stop| Main

    %% Keyboard interactions
    Key -->|stop_all_tx - global stop| Main
    Key -->|stop_play_tx - ESC stop| Play

    %% Daemon mode interactions
    Ctl -->|recording_paused / utterance_meta - push-to-talk with selection| Rec
    Ctl -->|tx_tts - read selection aloud| TTS
    Ctl -->|stop_play_tx + interrupt_counter - stop speech| Play
    Conv -->|tx_action - paste transcript at cursor| Ctl
    UI -.->|tx_ui lines| Sink
    Sink -->|Ui / State / History messages| Srv
    Srv -->|snapshot + live stream| Att
    Att -->|Key events| Srv
    Srv -->|ClientEvent - keys, Say| Ctl
